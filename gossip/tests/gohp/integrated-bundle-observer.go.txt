package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"

	rt "github.com/katsuakikureyama/GoHP/runtime"
)

type integratedBundleCase struct {
	Name      string `json:"name"`
	Operation string `json:"operation"`
	Args      []any  `json:"args"`
}

type integratedBundleObservation struct {
	Name  string `json:"name"`
	Type  string `json:"type"`
	Value any    `json:"value"`
}

func integratedBundleFixtureValue(value any) any {
	switch typed := value.(type) {
	case nil, bool, string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil {
			panic(fmt.Sprintf("integrated bundle only transports modest signed integer JSON numbers: %v", err))
		}
		return integer
	case []any:
		entries := make([]rt.Entry, 0, len(typed))
		for _, item := range typed {
			entries = append(entries, rt.Entry{Value: integratedBundleFixtureValue(item)})
		}
		return rt.NewArray(entries...)
	case map[string]any:
		panic("integrated bundle does not transport PHP objects/maps")
	default:
		panic(fmt.Sprintf("integrated bundle unsupported JSON value type %T", value))
	}
}

func integratedBundleType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	case rt.Array:
		return "array"
	default:
		panic(fmt.Sprintf("integrated bundle unexpected translated value type %T", value))
	}
}

func normalizeIntegratedBundleValue(value any) any {
	switch typed := value.(type) {
	case nil, bool, int64, string:
		return typed
	case rt.Array:
		out := make([]any, len(typed.Entries))
		for index, entry := range typed.Entries {
			key, ok := entry.Key.(int64)
			if !ok || key != int64(index) {
				panic("integrated bundle output array must retain sequential integer keys")
			}
			out[index] = normalizeIntegratedBundleValue(entry.Value)
		}
		return out
	default:
		panic(fmt.Sprintf("integrated bundle cannot normalize translated output type %T", value))
	}
}

func integratedBundleNoArgs(args []any, operation string) {
	if len(args) != 0 {
		panic(fmt.Sprintf("%s requires no arguments", operation))
	}
}

func integratedBundleString(args []any, operation string) string {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one string argument", operation))
	}
	value, ok := args[0].(string)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly one string argument", operation))
	}
	return value
}

func integratedBundleWidth(value any, operation string) int64 {
	width, ok := value.(int64)
	if !ok || width < 0 || width > 4096 {
		panic(fmt.Sprintf("%s width must be an int in 0..4096", operation))
	}
	return width
}

func integratedBundleValueAndWidth(args []any, operation string) (any, int64) {
	if len(args) != 2 {
		panic(fmt.Sprintf("%s requires a string or int value and int width", operation))
	}
	switch args[0].(type) {
	case string, int64:
		return args[0], integratedBundleWidth(args[1], operation)
	default:
		panic(fmt.Sprintf("%s requires a string or int value and int width", operation))
	}
}

func integratedBundleCaptures(args []any, operation string) rt.Array {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires one non-empty list of string captures", operation))
	}
	captures, ok := args[0].(rt.Array)
	if !ok || len(captures.Entries) == 0 {
		panic(fmt.Sprintf("%s requires one non-empty list of string captures", operation))
	}
	for index, entry := range captures.Entries {
		key, keyOK := entry.Key.(int64)
		if !keyOK || key != int64(index) {
			panic(fmt.Sprintf("%s captures must retain sequential integer keys", operation))
		}
		if _, ok := entry.Value.(string); !ok {
			panic(fmt.Sprintf("%s captures must all be strings", operation))
		}
	}
	return captures
}

func integratedBundleArray(args []any, operation string) rt.Array {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one list array", operation))
	}
	value, ok := args[0].(rt.Array)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly one list array", operation))
	}
	return value
}

func integratedBundleCall(testCase integratedBundleCase, args []any) any {
	switch testCase.Operation {
	case "__return_true":
		integratedBundleNoArgs(args, testCase.Operation)
		return F___return_true()
	case "__return_false":
		integratedBundleNoArgs(args, testCase.Operation)
		return F___return_false()
	case "__return_zero":
		integratedBundleNoArgs(args, testCase.Operation)
		return F___return_zero()
	case "__return_empty_array":
		integratedBundleNoArgs(args, testCase.Operation)
		return F___return_empty_array()
	case "__return_null":
		integratedBundleNoArgs(args, testCase.Operation)
		return F___return_null()
	case "__return_empty_string":
		integratedBundleNoArgs(args, testCase.Operation)
		return F___return_empty_string()
	case "wp_validate_boolean":
		return F_wp_validate_boolean(integratedBundleString(args, testCase.Operation))
	case "_autop_newline_preservation_helper":
		return F__autop_newline_preservation_helper(integratedBundleCaptures(args, testCase.Operation))
	case "zeroise":
		value, width := integratedBundleValueAndWidth(args, testCase.Operation)
		return F_zeroise(value, width)
	case "trailingslashit":
		return F_trailingslashit(integratedBundleString(args, testCase.Operation))
	case "untrailingslashit":
		return F_untrailingslashit(integratedBundleString(args, testCase.Operation))
	case "stripslashes_from_strings_only":
		if len(args) != 1 {
			panic("stripslashes_from_strings_only requires exactly one fixture value")
		}
		return F_stripslashes_from_strings_only(args[0])
	case "gossip_capture_newline_unslash":
		return F_gossip_capture_newline_unslash(integratedBundleCaptures(args, testCase.Operation))
	case "gossip_unslash_array_keys":
		return F_gossip_unslash_array_keys(integratedBundleArray(args, testCase.Operation))
	case "gossip_unslash_array_shape":
		return F_gossip_unslash_array_shape(integratedBundleArray(args, testCase.Operation))
	case "gossip_path_pad":
		value, width := integratedBundleValueAndWidth(args, testCase.Operation)
		path, ok := value.(string)
		if !ok {
			panic("gossip_path_pad requires a string path")
		}
		return F_gossip_path_pad(path, width)
	case "gossip_zero_pad":
		if len(args) != 1 {
			panic("gossip_zero_pad requires one width")
		}
		return F_gossip_zero_pad(integratedBundleWidth(args[0], testCase.Operation))
	case "gossip_false_boolean":
		integratedBundleNoArgs(args, testCase.Operation)
		return F_gossip_false_boolean()
	case "gossip_empty_string_path":
		integratedBundleNoArgs(args, testCase.Operation)
		return F_gossip_empty_string_path()
	default:
		panic(fmt.Sprintf("unknown operation: %s", testCase.Operation))
	}
}

// init is compiled beside GoHP's generated main.go. It dispatches only to
// exact generated functions and normalizes typed runtime values for JSON.
func init() {
	data, err := os.ReadFile("/input/cases.json")
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	var cases []integratedBundleCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic("integrated bundle fixture has trailing JSON values")
	}

	observations := make([]integratedBundleObservation, 0, len(cases))
	for _, testCase := range cases {
		if testCase.Name == "" || testCase.Operation == "" {
			panic("integrated bundle case needs nonempty name and operation")
		}
		args := make([]any, len(testCase.Args))
		for index, arg := range testCase.Args {
			args[index] = integratedBundleFixtureValue(arg)
		}
		value := integratedBundleCall(testCase, args)
		observations = append(observations, integratedBundleObservation{
			Name: testCase.Name, Type: integratedBundleType(value), Value: normalizeIntegratedBundleValue(value),
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
