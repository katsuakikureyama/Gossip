package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"

	rt "github.com/katsuakikureyama/GoHP/runtime"
)

type recursiveBundleCase struct {
	Name      string `json:"name"`
	Operation string `json:"operation"`
	Args      []any  `json:"args"`
}

type recursiveBundleObservation struct {
	Name      string `json:"name"`
	Type      string `json:"type"`
	Value     any    `json:"value"`
	ArgsAfter []any  `json:"args_after"`
}

func recursiveBundleFixtureValue(value any) any {
	switch typed := value.(type) {
	case nil, bool, string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil {
			panic(fmt.Sprintf("recursive bundle only transports modest signed integer JSON numbers: %v", err))
		}
		return integer
	case []any:
		entries := make([]rt.Entry, 0, len(typed))
		for _, item := range typed {
			entries = append(entries, rt.Entry{Value: recursiveBundleFixtureValue(item)})
		}
		return rt.NewArray(entries...)
	case map[string]any:
		panic("recursive bundle does not transport PHP objects/maps")
	default:
		panic(fmt.Sprintf("recursive bundle unsupported JSON value type %T", value))
	}
}

func recursiveBundleType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	case rt.Array:
		return "array"
	default:
		panic(fmt.Sprintf("recursive bundle unexpected translated value type %T", value))
	}
}

func normalizeRecursiveBundleValue(value any) any {
	switch typed := value.(type) {
	case nil, bool, int64, string:
		return typed
	case rt.Array:
		out := make([]any, len(typed.Entries))
		for index, entry := range typed.Entries {
			key, ok := entry.Key.(int64)
			if !ok || key != int64(index) {
				panic("recursive bundle output array must retain sequential integer keys")
			}
			out[index] = normalizeRecursiveBundleValue(entry.Value)
		}
		return out
	default:
		panic(fmt.Sprintf("recursive bundle cannot normalize translated output type %T", value))
	}
}

func recursiveBundleNoArgs(args []any, operation string) {
	if len(args) != 0 {
		panic(fmt.Sprintf("%s requires no arguments", operation))
	}
}

func recursiveBundleString(args []any, operation string) string {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one string argument", operation))
	}
	value, ok := args[0].(string)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly one string argument", operation))
	}
	return value
}

func recursiveBundleAny(args []any, operation string) any {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one fixture value", operation))
	}
	switch args[0].(type) {
	case nil, bool, int64, string, rt.Array:
		return args[0]
	default:
		panic(fmt.Sprintf("%s requires exactly one fixture value", operation))
	}
}

func recursiveBundleWidth(value any, operation string) int64 {
	width, ok := value.(int64)
	if !ok || width < 0 || width > 4096 {
		panic(fmt.Sprintf("%s width must be an int in 0..4096", operation))
	}
	return width
}

func recursiveBundleValueAndWidth(args []any, operation string) (any, int64) {
	if len(args) != 2 {
		panic(fmt.Sprintf("%s requires a string or int value and int width", operation))
	}
	switch args[0].(type) {
	case string, int64:
		return args[0], recursiveBundleWidth(args[1], operation)
	default:
		panic(fmt.Sprintf("%s requires a string or int value and int width", operation))
	}
}

func recursiveBundleCaptures(args []any, operation string) rt.Array {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires one non-empty list of string captures", operation))
	}
	captures, ok := args[0].(rt.Array)
	if !ok || len(captures.Entries) == 0 {
		panic(fmt.Sprintf("%s requires one non-empty list of string captures", operation))
	}
	for index, entry := range captures.Entries {
		key, keyOK := entry.Key.(int64)
		if !keyOK || key != int64(index) {
			panic(fmt.Sprintf("%s captures must retain sequential integer keys", operation))
		}
		if _, ok := entry.Value.(string); !ok {
			panic(fmt.Sprintf("%s captures must all be strings", operation))
		}
	}
	return captures
}

func recursiveBundleArray(args []any, operation string) rt.Array {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one list array", operation))
	}
	value, ok := args[0].(rt.Array)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly one list array", operation))
	}
	return value
}

func recursiveBundleCall(testCase recursiveBundleCase, args []any) any {
	switch testCase.Operation {
	case "__return_true":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F___return_true()
	case "__return_false":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F___return_false()
	case "__return_zero":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F___return_zero()
	case "__return_empty_array":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F___return_empty_array()
	case "__return_null":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F___return_null()
	case "__return_empty_string":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F___return_empty_string()
	case "wp_validate_boolean":
		return F_wp_validate_boolean(recursiveBundleString(args, testCase.Operation))
	case "_autop_newline_preservation_helper":
		return F__autop_newline_preservation_helper(recursiveBundleCaptures(args, testCase.Operation))
	case "zeroise":
		value, width := recursiveBundleValueAndWidth(args, testCase.Operation)
		return F_zeroise(value, width)
	case "trailingslashit":
		return F_trailingslashit(recursiveBundleString(args, testCase.Operation))
	case "untrailingslashit":
		return F_untrailingslashit(recursiveBundleString(args, testCase.Operation))
	case "stripslashes_from_strings_only":
		return F_stripslashes_from_strings_only(recursiveBundleAny(args, testCase.Operation))
	case "wp_slash":
		return F_wp_slash(recursiveBundleAny(args, testCase.Operation))
	case "gossip_capture_newline_unslash":
		return F_gossip_capture_newline_unslash(recursiveBundleCaptures(args, testCase.Operation))
	case "gossip_unslash_array_keys":
		return F_gossip_unslash_array_keys(recursiveBundleArray(args, testCase.Operation))
	case "gossip_unslash_array_shape":
		return F_gossip_unslash_array_shape(recursiveBundleArray(args, testCase.Operation))
	case "gossip_path_pad":
		value, width := recursiveBundleValueAndWidth(args, testCase.Operation)
		path, ok := value.(string)
		if !ok {
			panic("gossip_path_pad requires a string path")
		}
		return F_gossip_path_pad(path, width)
	case "gossip_zero_pad":
		if len(args) != 1 {
			panic("gossip_zero_pad requires one width")
		}
		return F_gossip_zero_pad(recursiveBundleWidth(args[0], testCase.Operation))
	case "gossip_false_boolean":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F_gossip_false_boolean()
	case "gossip_empty_string_path":
		recursiveBundleNoArgs(args, testCase.Operation)
		return F_gossip_empty_string_path()
	case "gossip_slash_scalar_roundtrip":
		return F_gossip_slash_scalar_roundtrip(recursiveBundleString(args, testCase.Operation))
	case "gossip_slash_list_shallow_roundtrip":
		return F_gossip_slash_list_shallow_roundtrip(recursiveBundleArray(args, testCase.Operation))
	case "gossip_slash_capture_newline_unslash":
		return F_gossip_slash_capture_newline_unslash(recursiveBundleCaptures(args, testCase.Operation))
	case "gossip_slash_array_shape":
		return F_gossip_slash_array_shape(recursiveBundleArray(args, testCase.Operation))
	default:
		panic(fmt.Sprintf("unknown operation: %s", testCase.Operation))
	}
}

// init is compiled beside GoHP's generated main.go. It dispatches only to
// exact generated functions and observes the caller arguments after each call.
func init() {
	data, err := os.ReadFile("/input/cases.json")
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	var cases []recursiveBundleCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic("recursive bundle fixture has trailing JSON values")
	}

	observations := make([]recursiveBundleObservation, 0, len(cases))
	for _, testCase := range cases {
		if testCase.Name == "" || testCase.Operation == "" {
			panic("recursive bundle case needs nonempty name and operation")
		}
		args := make([]any, len(testCase.Args))
		for index, arg := range testCase.Args {
			args[index] = recursiveBundleFixtureValue(arg)
		}
		value := recursiveBundleCall(testCase, args)
		argsAfter := make([]any, len(args))
		for index, arg := range args {
			argsAfter[index] = normalizeRecursiveBundleValue(arg)
		}
		observations = append(observations, recursiveBundleObservation{
			Name: testCase.Name, Type: recursiveBundleType(value), Value: normalizeRecursiveBundleValue(value), ArgsAfter: argsAfter,
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
