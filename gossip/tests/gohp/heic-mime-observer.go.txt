package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"
)

type heicMimeCase struct {
	Name      string `json:"name"`
	InputType string `json:"input_type"`
	Input     any    `json:"input"`
	Expected  bool   `json:"expected"`
}

type heicMimeValueObservation struct {
	Type  string `json:"type"`
	Value any    `json:"value"`
}

type heicMimeObservation struct {
	Name       string                   `json:"name"`
	InputType  string                   `json:"input_type"`
	Type       string                   `json:"type"`
	Value      bool                     `json:"value"`
	InputAfter heicMimeValueObservation `json:"input_after"`
}

func heicMimeFixtureInput(value any) any {
	switch typed := value.(type) {
	case nil, bool, string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil || integer < -1_000_000 || integer > 1_000_000 {
			panic(fmt.Sprintf("HEIC MIME fixture input must be a modest signed integer: %v", value))
		}
		return integer
	case []any, map[string]any:
		panic("HEIC MIME fixture transports scalar labels only, not arrays or objects")
	default:
		panic(fmt.Sprintf("HEIC MIME fixture unsupported JSON input type %T", value))
	}
}

func heicMimeFixtureType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	default:
		panic(fmt.Sprintf("HEIC MIME fixture unexpected translated value type %T", value))
	}
}

// init is compiled beside GoHP's generated main.go. It invokes only the
// generated original helper; expected values remain host-side assertions.
func init() {
	data, err := os.ReadFile("/input/cases.json")
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	var cases []heicMimeCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic("HEIC MIME fixture has trailing JSON values")
	}

	observations := make([]heicMimeObservation, 0, len(cases))
	for _, testCase := range cases {
		if testCase.Name == "" {
			panic("HEIC MIME fixture case name is required")
		}
		input := heicMimeFixtureInput(testCase.Input)
		if actualType := heicMimeFixtureType(input); actualType != testCase.InputType {
			panic(fmt.Sprintf("case input type did not survive JSON transport: %s: got %s want %s", testCase.Name, actualType, testCase.InputType))
		}
		value, ok := F_wp_is_heic_image_mime_type(input).(bool)
		if !ok {
			panic("wp_is_heic_image_mime_type did not return bool")
		}
		observations = append(observations, heicMimeObservation{
			Name: testCase.Name, InputType: testCase.InputType, Type: "bool", Value: value,
			InputAfter: heicMimeValueObservation{Type: heicMimeFixtureType(input), Value: input},
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
