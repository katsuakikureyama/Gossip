package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"os"

	rt "github.com/katsuakikureyama/GoHP/runtime"
)

type booleanCase struct {
	Name      string `json:"name"`
	InputType string `json:"input_type"`
	Input     any    `json:"input"`
}

type booleanObservation struct {
	Name      string `json:"name"`
	InputType string `json:"input_type"`
	Type      string `json:"type"`
	Value     bool   `json:"value"`
}

func booleanInput(value any) any {
	switch typed := value.(type) {
	case nil, bool, string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil {
			panic(fmt.Sprintf("boolean fixture only transports signed integer JSON numbers: %v", err))
		}
		return integer
	case []any:
		entries := make([]rt.Entry, 0, len(typed))
		for _, item := range typed {
			entries = append(entries, rt.Entry{Value: booleanInput(item)})
		}
		return rt.NewArray(entries...)
	case map[string]any:
		panic("boolean fixture does not transport PHP objects/maps")
	default:
		panic(fmt.Sprintf("boolean fixture unsupported JSON input type %T", value))
	}
}

func booleanType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	case rt.Array:
		return "array"
	default:
		panic(fmt.Sprintf("boolean fixture unexpected translated input type %T", value))
	}
}

// init is compiled beside GoHP's generated main.go. It observes calls to the
// exact generated function, while preserving array values in the runtime type.
func init() {
	data, err := os.ReadFile("/input/cases.json")
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	var cases []booleanCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	if decoder.More() {
		panic("boolean fixture has trailing JSON values")
	}

	observations := make([]booleanObservation, 0, len(cases))
	for _, testCase := range cases {
		input := booleanInput(testCase.Input)
		if actualType := booleanType(input); actualType != testCase.InputType {
			panic(fmt.Sprintf("case input type did not survive JSON transport: %s: got %s want %s", testCase.Name, actualType, testCase.InputType))
		}
		value, ok := F_wp_validate_boolean(input).(bool)
		if !ok {
			panic("wp_validate_boolean did not return bool")
		}
		observations = append(observations, booleanObservation{
			Name: testCase.Name, InputType: testCase.InputType, Type: "bool", Value: value,
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
