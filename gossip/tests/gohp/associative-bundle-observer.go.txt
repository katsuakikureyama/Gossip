package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"

	rt "github.com/katsuakikureyama/GoHP/runtime"
)

type associativeBundleCase struct {
	Name      string `json:"name"`
	Operation string `json:"operation"`
	Args      []any  `json:"args"`
}

type associativeBundleObservation struct {
	Name      string `json:"name"`
	Type      string `json:"type"`
	Value     any    `json:"value"`
	ArgsAfter []any  `json:"args_after"`
}

func associativeBundleFixtureKey(value any) any {
	switch typed := value.(type) {
	case string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil || integer < -1_000_000 || integer > 1_000_000 {
			panic(fmt.Sprintf("associative bundle fixture key must be a bounded signed integer or string: %v", value))
		}
		return integer
	default:
		panic(fmt.Sprintf("associative bundle fixture key must be a signed integer or string, got %T", value))
	}
}

func associativeBundleFixtureValue(value any, depth int) any {
	if depth > 8 {
		panic("associative bundle fixture values are limited to depth eight")
	}
	switch typed := value.(type) {
	case nil, bool, string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil || integer < -1_000_000 || integer > 1_000_000 {
			panic(fmt.Sprintf("associative bundle only transports modest signed integer JSON numbers: %v", err))
		}
		return integer
	case []any:
		entries := make([]rt.Entry, 0, len(typed))
		for _, item := range typed {
			entries = append(entries, rt.Entry{Value: associativeBundleFixtureValue(item, depth+1)})
		}
		return rt.NewArray(entries...)
	case map[string]any:
		panic("associative bundle does not transport PHP objects/maps")
	default:
		panic(fmt.Sprintf("associative bundle unsupported JSON value type %T", value))
	}
}

func associativeBundleFixturePairs(value []any, field string) rt.Array {
	entries := make([]rt.Entry, 0, len(value))
	for index, rawPair := range value {
		pair, ok := rawPair.([]any)
		if !ok || len(pair) != 2 {
			panic(fmt.Sprintf("associative bundle fixture %s pair %d must contain exactly key and value", field, index))
		}
		entries = append(entries, rt.Entry{
			Key: associativeBundleFixtureKey(pair[0]), Value: associativeBundleFixtureValue(pair[1], 0),
		})
	}
	return rt.NewArray(entries...)
}

func associativeBundleFixtureSelectors(value []any, operation string) rt.Array {
	entries := make([]rt.Entry, 0, len(value))
	for _, key := range value {
		entries = append(entries, rt.Entry{Value: associativeBundleFixtureKey(key)})
	}
	return rt.NewArray(entries...)
}

func associativeBundleType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	case rt.Array:
		return "array"
	default:
		panic(fmt.Sprintf("associative bundle unexpected translated value type %T", value))
	}
}

func normalizeAssociativeBundleValue(value any, depth int) any {
	if depth > 8 {
		panic("associative bundle fixture output exceeds depth eight")
	}
	switch typed := value.(type) {
	case nil, bool, int64, string:
		return typed
	case rt.Array:
		out := make([]any, len(typed.Entries))
		for index, entry := range typed.Entries {
			key, ok := entry.Key.(int64)
			if !ok || key != int64(index) {
				panic("associative bundle output array must retain sequential integer keys")
			}
			out[index] = normalizeAssociativeBundleValue(entry.Value, depth+1)
		}
		return out
	default:
		panic(fmt.Sprintf("associative bundle cannot normalize translated output type %T", value))
	}
}

func normalizeAssociativeBundleEntries(value any) [][]any {
	array, ok := value.(rt.Array)
	if !ok {
		panic("associative bundle associative root must be an array")
	}
	out := make([][]any, 0, len(array.Entries))
	for _, entry := range array.Entries {
		switch key := entry.Key.(type) {
		case int64:
			if key < -1_000_000 || key > 1_000_000 {
				panic("associative bundle output integer key is outside the fixture bounds")
			}
		case string:
		default:
			panic(fmt.Sprintf("associative bundle output has unsupported key type %T", entry.Key))
		}
		out = append(out, []any{entry.Key, normalizeAssociativeBundleValue(entry.Value, 0)})
	}
	return out
}

func associativeBundlePairOperation(operation string) bool {
	switch operation {
	case "wp_array_slice_assoc", "gossip_slice_slash", "gossip_slash_slice",
		"gossip_slice_shallow_roundtrip", "gossip_slice_shape", "gossip_slice_boolean":
		return true
	default:
		return false
	}
}

func associativeBundlePairsResult(operation string) bool {
	switch operation {
	case "wp_array_slice_assoc", "gossip_slice_slash", "gossip_slash_slice", "gossip_slice_shallow_roundtrip":
		return true
	default:
		return false
	}
}

func associativeBundlePairArgs(args []any, operation string) ([]any, rt.Array, rt.Array) {
	if len(args) != 2 {
		panic(fmt.Sprintf("%s requires exactly an ordered-pairs array and a selector list", operation))
	}
	rawPairs, ok := args[0].([]any)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly an ordered-pairs array and a selector list", operation))
	}
	rawSelectors, ok := args[1].([]any)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly an ordered-pairs array and a selector list", operation))
	}
	input := associativeBundleFixturePairs(rawPairs, operation+" input")
	selectors := associativeBundleFixtureSelectors(rawSelectors, operation)
	return []any{input, selectors}, input, selectors
}

func associativeBundlePairRuntimeArgs(args []any, operation string) (rt.Array, rt.Array) {
	if len(args) != 2 {
		panic(fmt.Sprintf("%s requires exactly an ordered-pairs array and a selector list", operation))
	}
	input, inputOK := args[0].(rt.Array)
	selectors, selectorsOK := args[1].(rt.Array)
	if !inputOK || !selectorsOK {
		panic(fmt.Sprintf("%s requires exactly an ordered-pairs array and a selector list", operation))
	}
	return input, selectors
}

func associativeBundleFixtureArgs(testCase associativeBundleCase) []any {
	if associativeBundlePairOperation(testCase.Operation) {
		args, _, _ := associativeBundlePairArgs(testCase.Args, testCase.Operation)
		return args
	}
	args := make([]any, len(testCase.Args))
	for index, arg := range testCase.Args {
		args[index] = associativeBundleFixtureValue(arg, 0)
	}
	return args
}

func associativeBundleObservedValue(operation string, value any) any {
	if associativeBundlePairsResult(operation) {
		return normalizeAssociativeBundleEntries(value)
	}
	return normalizeAssociativeBundleValue(value, 0)
}

func associativeBundleObservedArgs(operation string, args []any) []any {
	out := make([]any, len(args))
	for index, arg := range args {
		if associativeBundlePairOperation(operation) && index == 0 {
			out[index] = normalizeAssociativeBundleEntries(arg)
			continue
		}
		out[index] = normalizeAssociativeBundleValue(arg, 0)
	}
	return out
}

func associativeBundleNoArgs(args []any, operation string) {
	if len(args) != 0 {
		panic(fmt.Sprintf("%s requires no arguments", operation))
	}
}

func associativeBundleString(args []any, operation string) string {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one string argument", operation))
	}
	value, ok := args[0].(string)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly one string argument", operation))
	}
	return value
}

func associativeBundleAny(args []any, operation string) any {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one fixture value", operation))
	}
	switch args[0].(type) {
	case nil, bool, int64, string, rt.Array:
		return args[0]
	default:
		panic(fmt.Sprintf("%s requires exactly one fixture value", operation))
	}
}

func associativeBundleWidth(value any, operation string) int64 {
	width, ok := value.(int64)
	if !ok || width < 0 || width > 4096 {
		panic(fmt.Sprintf("%s width must be an int in 0..4096", operation))
	}
	return width
}

func associativeBundleValueAndWidth(args []any, operation string) (any, int64) {
	if len(args) != 2 {
		panic(fmt.Sprintf("%s requires a string or int value and int width", operation))
	}
	switch args[0].(type) {
	case string, int64:
		return args[0], associativeBundleWidth(args[1], operation)
	default:
		panic(fmt.Sprintf("%s requires a string or int value and int width", operation))
	}
}

func associativeBundleCaptures(args []any, operation string) rt.Array {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires one non-empty list of string captures", operation))
	}
	captures, ok := args[0].(rt.Array)
	if !ok || len(captures.Entries) == 0 {
		panic(fmt.Sprintf("%s requires one non-empty list of string captures", operation))
	}
	for index, entry := range captures.Entries {
		key, keyOK := entry.Key.(int64)
		if !keyOK || key != int64(index) {
			panic(fmt.Sprintf("%s captures must retain sequential integer keys", operation))
		}
		if _, ok := entry.Value.(string); !ok {
			panic(fmt.Sprintf("%s captures must all be strings", operation))
		}
	}
	return captures
}

func associativeBundleArray(args []any, operation string) rt.Array {
	if len(args) != 1 {
		panic(fmt.Sprintf("%s requires exactly one list array", operation))
	}
	value, ok := args[0].(rt.Array)
	if !ok {
		panic(fmt.Sprintf("%s requires exactly one list array", operation))
	}
	return value
}

func associativeBundleCall(testCase associativeBundleCase, args []any) any {
	switch testCase.Operation {
	case "__return_true":
		associativeBundleNoArgs(args, testCase.Operation)
		return F___return_true()
	case "__return_false":
		associativeBundleNoArgs(args, testCase.Operation)
		return F___return_false()
	case "__return_zero":
		associativeBundleNoArgs(args, testCase.Operation)
		return F___return_zero()
	case "__return_empty_array":
		associativeBundleNoArgs(args, testCase.Operation)
		return F___return_empty_array()
	case "__return_null":
		associativeBundleNoArgs(args, testCase.Operation)
		return F___return_null()
	case "__return_empty_string":
		associativeBundleNoArgs(args, testCase.Operation)
		return F___return_empty_string()
	case "wp_validate_boolean":
		return F_wp_validate_boolean(associativeBundleString(args, testCase.Operation))
	case "_autop_newline_preservation_helper":
		return F__autop_newline_preservation_helper(associativeBundleCaptures(args, testCase.Operation))
	case "zeroise":
		value, width := associativeBundleValueAndWidth(args, testCase.Operation)
		return F_zeroise(value, width)
	case "trailingslashit":
		return F_trailingslashit(associativeBundleString(args, testCase.Operation))
	case "untrailingslashit":
		return F_untrailingslashit(associativeBundleString(args, testCase.Operation))
	case "stripslashes_from_strings_only":
		return F_stripslashes_from_strings_only(associativeBundleAny(args, testCase.Operation))
	case "wp_slash":
		return F_wp_slash(associativeBundleAny(args, testCase.Operation))
	case "gossip_capture_newline_unslash":
		return F_gossip_capture_newline_unslash(associativeBundleCaptures(args, testCase.Operation))
	case "gossip_unslash_array_keys":
		return F_gossip_unslash_array_keys(associativeBundleArray(args, testCase.Operation))
	case "gossip_unslash_array_shape":
		return F_gossip_unslash_array_shape(associativeBundleArray(args, testCase.Operation))
	case "gossip_path_pad":
		value, width := associativeBundleValueAndWidth(args, testCase.Operation)
		path, ok := value.(string)
		if !ok {
			panic("gossip_path_pad requires a string path")
		}
		return F_gossip_path_pad(path, width)
	case "gossip_zero_pad":
		if len(args) != 1 {
			panic("gossip_zero_pad requires one width")
		}
		return F_gossip_zero_pad(associativeBundleWidth(args[0], testCase.Operation))
	case "gossip_false_boolean":
		associativeBundleNoArgs(args, testCase.Operation)
		return F_gossip_false_boolean()
	case "gossip_empty_string_path":
		associativeBundleNoArgs(args, testCase.Operation)
		return F_gossip_empty_string_path()
	case "gossip_slash_scalar_roundtrip":
		return F_gossip_slash_scalar_roundtrip(associativeBundleString(args, testCase.Operation))
	case "gossip_slash_list_shallow_roundtrip":
		return F_gossip_slash_list_shallow_roundtrip(associativeBundleArray(args, testCase.Operation))
	case "gossip_slash_capture_newline_unslash":
		return F_gossip_slash_capture_newline_unslash(associativeBundleCaptures(args, testCase.Operation))
	case "gossip_slash_array_shape":
		return F_gossip_slash_array_shape(associativeBundleArray(args, testCase.Operation))
	case "wp_array_slice_assoc":
		input, selectors := associativeBundlePairRuntimeArgs(args, testCase.Operation)
		return F_wp_array_slice_assoc(input, selectors)
	case "gossip_slice_slash":
		input, selectors := associativeBundlePairRuntimeArgs(args, testCase.Operation)
		return F_gossip_slice_slash(input, selectors)
	case "gossip_slash_slice":
		input, selectors := associativeBundlePairRuntimeArgs(args, testCase.Operation)
		return F_gossip_slash_slice(input, selectors)
	case "gossip_slice_shallow_roundtrip":
		input, selectors := associativeBundlePairRuntimeArgs(args, testCase.Operation)
		return F_gossip_slice_shallow_roundtrip(input, selectors)
	case "gossip_slice_shape":
		input, selectors := associativeBundlePairRuntimeArgs(args, testCase.Operation)
		return F_gossip_slice_shape(input, selectors)
	case "gossip_slice_boolean":
		input, selectors := associativeBundlePairRuntimeArgs(args, testCase.Operation)
		return F_gossip_slice_boolean(input, selectors)
	default:
		panic(fmt.Sprintf("unknown operation: %s", testCase.Operation))
	}
}

// init is compiled beside GoHP's generated main.go. It dispatches only to
// exact generated functions and observes the caller arguments after each call.
func init() {
	data, err := os.ReadFile("/input/cases.json")
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	var cases []associativeBundleCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic("associative bundle fixture has trailing JSON values")
	}

	observations := make([]associativeBundleObservation, 0, len(cases))
	for _, testCase := range cases {
		if testCase.Name == "" || testCase.Operation == "" {
			panic("associative bundle case needs nonempty name and operation")
		}
		args := associativeBundleFixtureArgs(testCase)
		value := associativeBundleCall(testCase, args)
		observations = append(observations, associativeBundleObservation{
			Name: testCase.Name, Type: associativeBundleType(value),
			Value:     associativeBundleObservedValue(testCase.Operation, value),
			ArgsAfter: associativeBundleObservedArgs(testCase.Operation, args),
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
