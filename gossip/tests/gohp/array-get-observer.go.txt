package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"
	"reflect"

	rt "github.com/katsuakikureyama/GoHP/runtime"
)

const (
	arrayGetMaxDepth   = 12
	arrayGetMaxEntries = 256
	arrayGetMaxString  = 4096
	arrayGetMaxInteger = 1_000_000
)

type arrayGetCase struct {
	Name        string          `json:"name"`
	Input       json.RawMessage `json:"input"`
	Path        json.RawMessage `json:"path"`
	Default     json.RawMessage `json:"default"`
	Expected    json.RawMessage `json:"expected"`
	OmitDefault *bool           `json:"omit_default"`
}

type arrayGetObservation struct {
	Name         string `json:"name"`
	Type         string `json:"type"`
	Value        any    `json:"value"`
	InputAfter   any    `json:"input_after"`
	PathAfter    any    `json:"path_after"`
	DefaultAfter any    `json:"default_after"`
}

func arrayGetDecodeCases(path string) []arrayGetCase {
	data, err := os.ReadFile(path)
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	decoder.DisallowUnknownFields()
	var cases []arrayGetCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic("array-get cases have trailing JSON values")
	}
	if cases == nil {
		panic("array-get cases must be an ordered list")
	}
	return cases
}

func arrayGetDecodeRaw(raw json.RawMessage, field string) any {
	if raw == nil {
		panic(field + " is required")
	}
	decoder := json.NewDecoder(bytes.NewReader(raw))
	decoder.UseNumber()
	var node any
	if err := decoder.Decode(&node); err != nil {
		panic(fmt.Sprintf("%s is not valid JSON: %v", field, err))
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic(field + " has trailing JSON values")
	}
	return node
}

func arrayGetInteger(value json.Number, field string) int64 {
	integer, err := value.Int64()
	if err != nil || integer < -arrayGetMaxInteger || integer > arrayGetMaxInteger {
		panic(fmt.Sprintf("%s integer is outside the bounded range", field))
	}
	return integer
}

func arrayGetKey(value any, field string) any {
	var raw any
	switch typed := value.(type) {
	case json.Number:
		raw = arrayGetInteger(typed, field+" key")
	case string:
		if len(typed) > arrayGetMaxString {
			panic(field + " string key exceeds the bounded size")
		}
		raw = typed
	default:
		panic(fmt.Sprintf("%s key must be int or string", field))
	}
	normalized := rt.Key(raw)
	if integer, ok := normalized.(int64); !ok || integer < -arrayGetMaxInteger || integer > arrayGetMaxInteger {
		if _, stringKey := normalized.(string); !stringKey {
			panic(fmt.Sprintf("%s normalized key has unsupported type %T", field, normalized))
		}
	}
	return normalized
}

func arrayGetDecodeNode(node any, field string, depth int, entries *int) any {
	if depth > arrayGetMaxDepth {
		panic(field + " exceeds the bounded depth")
	}
	switch typed := node.(type) {
	case nil, bool:
		return typed
	case json.Number:
		return arrayGetInteger(typed, field)
	case string:
		if len(typed) > arrayGetMaxString {
			panic(field + " string exceeds the bounded size")
		}
		return typed
	case map[string]any:
		if len(typed) != 1 {
			panic(field + " must be a scalar node or {array: ordered pairs}")
		}
		pairsRaw, ok := typed["array"]
		if !ok {
			panic(field + " must be a scalar node or {array: ordered pairs}")
		}
		pairs, ok := pairsRaw.([]any)
		if !ok {
			panic(field + " must be a scalar node or {array: ordered pairs}")
		}
		out := make([]rt.Entry, 0, len(pairs))
		seen := map[any]bool{}
		for index, rawPair := range pairs {
			pair, ok := rawPair.([]any)
			if !ok || len(pair) != 2 {
				panic(fmt.Sprintf("%s pair %d must contain exactly key and node", field, index))
			}
			key := arrayGetKey(pair[0], fmt.Sprintf("%s pair %d", field, index))
			if seen[key] {
				panic(fmt.Sprintf("%s has duplicate normalized key at pair %d", field, index))
			}
			seen[key] = true
			*entries++
			if *entries > arrayGetMaxEntries {
				panic(field + " exceeds the bounded entry count")
			}
			out = append(out, rt.Entry{Key: key, Value: arrayGetDecodeNode(pair[1], fmt.Sprintf("%s pair %d value", field, index), depth+1, entries)})
		}
		return rt.NewArray(out...)
	default:
		panic(fmt.Sprintf("%s must be a scalar node or {array: ordered pairs}", field))
	}
}

// arrayGetCanonicalNode validates the source transport without applying PHP
// key normalization. Comparing it with an encoded decoded value rejects a
// noncanonical expected array (for example both "7" and 7 as distinct keys).
func arrayGetCanonicalNode(node any, field string, depth int, entries *int) any {
	if depth > arrayGetMaxDepth {
		panic(field + " exceeds the bounded depth")
	}
	switch typed := node.(type) {
	case nil, bool:
		return typed
	case json.Number:
		return arrayGetInteger(typed, field)
	case string:
		if len(typed) > arrayGetMaxString {
			panic(field + " string exceeds the bounded size")
		}
		return typed
	case map[string]any:
		if len(typed) != 1 {
			panic(field + " must be a scalar node or {array: ordered pairs}")
		}
		pairsRaw, ok := typed["array"]
		if !ok {
			panic(field + " must be a scalar node or {array: ordered pairs}")
		}
		pairs, ok := pairsRaw.([]any)
		if !ok {
			panic(field + " must be a scalar node or {array: ordered pairs}")
		}
		out := make([]any, 0, len(pairs))
		for index, rawPair := range pairs {
			pair, ok := rawPair.([]any)
			if !ok || len(pair) != 2 {
				panic(fmt.Sprintf("%s pair %d must contain exactly key and node", field, index))
			}
			key := arrayGetCanonicalKey(pair[0], fmt.Sprintf("%s pair %d", field, index))
			*entries++
			if *entries > arrayGetMaxEntries {
				panic(field + " exceeds the bounded entry count")
			}
			out = append(out, []any{key, arrayGetCanonicalNode(pair[1], fmt.Sprintf("%s pair %d value", field, index), depth+1, entries)})
		}
		return map[string]any{"array": out}
	default:
		panic(fmt.Sprintf("%s must be a scalar node or {array: ordered pairs}", field))
	}
}

func arrayGetCanonicalKey(value any, field string) any {
	switch typed := value.(type) {
	case int64:
		if typed < -arrayGetMaxInteger || typed > arrayGetMaxInteger {
			panic(field + " integer key is outside the bounded range")
		}
		return typed
	case json.Number:
		return arrayGetInteger(typed, field+" key")
	case string:
		if len(typed) > arrayGetMaxString {
			panic(field + " string key exceeds the bounded size")
		}
		return typed
	default:
		panic(fmt.Sprintf("%s key must be int or string", field))
	}
}

func arrayGetType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	case rt.Array:
		return "array"
	default:
		panic(fmt.Sprintf("observed value has unsupported type %T", value))
	}
}

func arrayGetEncodeNode(value any, field string, depth int, entries *int) any {
	if depth > arrayGetMaxDepth {
		panic(field + " exceeds the bounded depth")
	}
	switch typed := value.(type) {
	case nil, bool:
		return typed
	case int64:
		if typed < -arrayGetMaxInteger || typed > arrayGetMaxInteger {
			panic(field + " integer is outside the bounded range")
		}
		return typed
	case string:
		if len(typed) > arrayGetMaxString {
			panic(field + " string exceeds the bounded size")
		}
		return typed
	case rt.Array:
		out := make([]any, 0, len(typed.Entries))
		for index, entry := range typed.Entries {
			key := arrayGetCanonicalKey(entry.Key, fmt.Sprintf("%s pair %d", field, index))
			*entries++
			if *entries > arrayGetMaxEntries {
				panic(field + " exceeds the bounded entry count")
			}
			out = append(out, []any{key, arrayGetEncodeNode(entry.Value, fmt.Sprintf("%s pair %d value", field, index), depth+1, entries)})
		}
		return map[string]any{"array": out}
	default:
		panic(fmt.Sprintf("%s has unsupported node type %T", field, value))
	}
}

func arrayGetObserve(value any, field string) any {
	entries := 0
	return arrayGetEncodeNode(value, field, 0, &entries)
}

// init is compiled beside GoHP's generated main.go. It invokes only the
// generated original helper; expected values assert the observation afterwards
// and never participate in the helper call.
func init() {
	cases := arrayGetDecodeCases("/input/cases.json")
	observations := make([]arrayGetObservation, 0, len(cases))
	seen := map[string]bool{}
	for _, testCase := range cases {
		if testCase.Name == "" || seen[testCase.Name] || testCase.OmitDefault == nil || testCase.Input == nil || testCase.Path == nil || testCase.Default == nil || testCase.Expected == nil {
			panic("invalid array-get case")
		}
		seen[testCase.Name] = true

		inputEntries, pathEntries, defaultEntries, expectedEntries := 0, 0, 0, 0
		input := arrayGetDecodeNode(arrayGetDecodeRaw(testCase.Input, testCase.Name+" input"), testCase.Name+" input", 0, &inputEntries)
		path := arrayGetDecodeNode(arrayGetDecodeRaw(testCase.Path, testCase.Name+" path"), testCase.Name+" path", 0, &pathEntries)
		defaultValue := arrayGetDecodeNode(arrayGetDecodeRaw(testCase.Default, testCase.Name+" default"), testCase.Name+" default", 0, &defaultEntries)
		expected := arrayGetDecodeNode(arrayGetDecodeRaw(testCase.Expected, testCase.Name+" expected"), testCase.Name+" expected", 0, &expectedEntries)

		expectedCanonicalEntries := 0
		expectedCanonical := arrayGetCanonicalNode(arrayGetDecodeRaw(testCase.Expected, testCase.Name+" expected"), testCase.Name+" expected", 0, &expectedCanonicalEntries)
		expectedNode := arrayGetObserve(expected, testCase.Name+" expected")
		if !reflect.DeepEqual(expectedNode, expectedCanonical) {
			panic("array-get expected node is not canonical: " + testCase.Name)
		}

		inputBefore, pathBefore, defaultBefore := rt.Clone(input), rt.Clone(path), rt.Clone(defaultValue)
		var value any
		if *testCase.OmitDefault {
			value = F__wp_array_get(input, path)
		} else {
			value = F__wp_array_get(input, path, defaultValue)
		}
		valueNode := arrayGetObserve(value, testCase.Name+" observed value")
		inputAfter := arrayGetObserve(input, testCase.Name+" input after")
		pathAfter := arrayGetObserve(path, testCase.Name+" path after")
		defaultAfter := arrayGetObserve(defaultValue, testCase.Name+" default after")
		if arrayGetType(value) != arrayGetType(expected) || !reflect.DeepEqual(valueNode, expectedNode) || !reflect.DeepEqual(input, inputBefore) || !reflect.DeepEqual(path, pathBefore) || !reflect.DeepEqual(defaultValue, defaultBefore) {
			panic("array-get expectation or mutation mismatch: " + testCase.Name)
		}

		observations = append(observations, arrayGetObservation{
			Name: testCase.Name, Type: arrayGetType(value), Value: valueNode,
			InputAfter: inputAfter, PathAfter: pathAfter, DefaultAfter: defaultAfter,
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
