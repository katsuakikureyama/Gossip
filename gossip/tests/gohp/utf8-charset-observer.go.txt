package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"

	rt "github.com/katsuakikureyama/GoHP/runtime"
)

type utf8CharsetCase struct {
	Name      string `json:"name"`
	InputType string `json:"input_type"`
	Input     any    `json:"input"`
	Expected  bool   `json:"expected"`
}

type utf8CharsetValueObservation struct {
	Type  string `json:"type"`
	Value any    `json:"value"`
}

type utf8CharsetObservation struct {
	Name       string                      `json:"name"`
	InputType  string                      `json:"input_type"`
	Type       string                      `json:"type"`
	Value      bool                        `json:"value"`
	InputAfter utf8CharsetValueObservation `json:"input_after"`
}

func utf8CharsetFixtureInput(value any, depth int) any {
	if depth > 8 {
		panic("UTF-8 charset fixture inputs are limited to depth eight")
	}
	switch typed := value.(type) {
	case nil, bool, string:
		return typed
	case json.Number:
		integer, err := typed.Int64()
		if err != nil || integer < -1_000_000 || integer > 1_000_000 {
			panic(fmt.Sprintf("UTF-8 charset fixture input must be a modest signed integer: %v", value))
		}
		return integer
	case []any:
		entries := make([]rt.Entry, 0, len(typed))
		for _, item := range typed {
			entries = append(entries, rt.Entry{Value: utf8CharsetFixtureInput(item, depth+1)})
		}
		return rt.NewArray(entries...)
	case map[string]any:
		panic("UTF-8 charset fixture does not transport PHP objects or maps")
	default:
		panic(fmt.Sprintf("UTF-8 charset fixture unsupported JSON input type %T", value))
	}
}

func utf8CharsetFixtureType(value any) string {
	switch value.(type) {
	case nil:
		return "null"
	case bool:
		return "bool"
	case int64:
		return "int"
	case string:
		return "string"
	case rt.Array:
		return "array"
	default:
		panic(fmt.Sprintf("UTF-8 charset fixture unexpected translated value type %T", value))
	}
}

func normalizeUTF8CharsetFixtureValue(value any, depth int) any {
	if depth > 8 {
		panic("UTF-8 charset fixture output exceeds depth eight")
	}
	switch typed := value.(type) {
	case nil, bool, int64, string:
		return typed
	case rt.Array:
		out := make([]any, len(typed.Entries))
		for index, entry := range typed.Entries {
			key, ok := entry.Key.(int64)
			if !ok || key != int64(index) {
				panic("UTF-8 charset fixture nested output must retain sequential integer keys")
			}
			out[index] = normalizeUTF8CharsetFixtureValue(entry.Value, depth+1)
		}
		return out
	default:
		panic(fmt.Sprintf("UTF-8 charset fixture cannot normalize output type %T", value))
	}
}

// init is compiled beside GoHP's generated main.go. It invokes only the
// generated original helper; expected values remain host-side assertions.
func init() {
	data, err := os.ReadFile("/input/cases.json")
	if err != nil {
		panic(err)
	}
	decoder := json.NewDecoder(bytes.NewReader(data))
	decoder.UseNumber()
	var cases []utf8CharsetCase
	if err := decoder.Decode(&cases); err != nil {
		panic(err)
	}
	var trailing any
	if err := decoder.Decode(&trailing); err != io.EOF {
		panic("UTF-8 charset fixture has trailing JSON values")
	}

	observations := make([]utf8CharsetObservation, 0, len(cases))
	for _, testCase := range cases {
		if testCase.Name == "" {
			panic("UTF-8 charset fixture case name is required")
		}
		input := utf8CharsetFixtureInput(testCase.Input, 0)
		if actualType := utf8CharsetFixtureType(input); actualType != testCase.InputType {
			panic(fmt.Sprintf("case input type did not survive JSON transport: %s: got %s want %s", testCase.Name, actualType, testCase.InputType))
		}
		value, ok := F__is_utf8_charset(input).(bool)
		if !ok {
			panic("_is_utf8_charset did not return bool")
		}
		observations = append(observations, utf8CharsetObservation{
			Name: testCase.Name, InputType: testCase.InputType, Type: "bool", Value: value,
			InputAfter: utf8CharsetValueObservation{Type: utf8CharsetFixtureType(input), Value: normalizeUTF8CharsetFixtureValue(input, 0)},
		})
	}
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
	os.Exit(0)
}
